# UX Polish Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Apply the six approved UX refinements (fixed bottom bar, dynamic title, dropdown section switcher, distinct bar, relocated deployment info, stable background) to the bottom-navigation web UI.

**Architecture:** Frontend-only changes to the static UI served by the Go `empirebusd` binary. The `navigation.js` router module and the hash/section state model are unchanged; `app.js` route application and action binding are adapted to the new markup, and `styles.css` gets the visual changes. HTML/CSS structure is asserted in Go tests; behaviour is verified with a jsdom harness against a local smoke server.

**Tech Stack:** Go (`cmd/empirebusd`, `service/api/httpapi` tests), vanilla HTML/CSS/JS, ESLint, Node `node:test` for the untouched router module, jsdom (dev-only, outside the repo) for interaction checks.

**Work location:** Commit directly on `main` in `/Users/rog/Development/xtura-automation` (matching the user's preference to push straight to main). Do not use the other worktrees.

- Do not change the hash routing contract: `#/overview`, `#/controls/water`, `#/more/tools`; defaults Controls→Heating, More→System; invalid hashes fall back to `#/overview`.
- Do not change section retention (`state.sections`) or any control widget, API call, safety behaviour, or SSE wiring.

```js
const { JSDOM, VirtualConsole } = require("jsdom");

const base = "http://127.0.0.1:18089";

async function load(hash) {
  const [index, navSrc, appSrc] = await Promise.all([
    fetch(`${base}/`).then((r) => r.text()),
    fetch(`${base}/static/navigation.js`).then((r) => r.text()),
    fetch(`${base}/static/app.js`).then((r) => r.text()),
  ]);
  const scripts = index.match(/<script[^>]*><\/script>/g) || [];
  let html = index.replace(/<script[^>]*><\/script>/g, () => "");
  html = html.replace("</body>", `${scripts[0] || ""}\n<script>${navSrc}</script>\n<script>${appSrc}</script>\n</body>`);

  const virtualConsole = new VirtualConsole();
  const errors = [];
  virtualConsole.on("jsdomError", (e) => errors.push(e.message));

  const dom = new JSDOM(html, {
    url: `${base}/${hash}`,
    runScripts: "dangerously",
    pretendToBeVisual: true,
    virtualConsole,
  });
  const { window } = dom;
  window.fetch = () => Promise.reject(new Error("network disabled"));
  window.EventSource = class {
    constructor() {}
    addEventListener() {}
    close() {}
  };
  window.dispatchEvent(new window.Event("DOMContentLoaded"));
  await new Promise((resolve) => setTimeout(resolve, 30));
  return { window, document: window.document, errors };
}

function visible(doc, id) {
  return !doc.getElementById(id).hidden;
}

async function waitFor(fn, label, timeoutMs = 1500) {
  const deadline = Date.now() + timeoutMs;
  while (Date.now() < deadline) {
    if (fn()) return;
    await new Promise((resolve) => setTimeout(resolve, 25));
  }
  throw new Error(`timeout waiting for ${label}`);
}

(async () => {
  const controls = await load("#/controls");
  if (controls.errors.length) {
    throw new Error(`script errors: ${controls.errors.join(" | ")}`);
  }
  const { document: doc, window: win } = controls;
  if (doc.title !== "Controls") throw new Error(`title=${doc.title}`);
  if (doc.getElementById("pageTitle").textContent !== "Controls") throw new Error("pageTitle not Controls");
  if (doc.getElementById("controlsSection").value !== "heating") throw new Error("default section not heating");

  const select = doc.getElementById("controlsSection");
  select.value = "water";
  select.dispatchEvent(new win.Event("change", { bubbles: true }));
  await waitFor(() => win.location.hash === "#/controls/water", "hash water");
  if (!visible(doc, "controlsWaterPanel")) throw new Error("water panel not visible");
  if (doc.title !== "Controls") throw new Error("title changed after section switch");

  doc.getElementById("locationNav").click();
  await waitFor(() => visible(doc, "locationPanel"), "location panel");
  if (doc.title !== "Location") throw new Error("title not Location");
  doc.getElementById("controlsNav").click();
  await waitFor(() => visible(doc, "controlsWaterPanel"), "water retained");
  if (doc.getElementById("controlsSection").value !== "water") throw new Error("section not retained");

  const tools = await load("#/more/tools");
  if (tools.window.document.title !== "More") throw new Error("title not More");
  if (tools.window.document.getElementById("moreSection").value !== "tools") throw new Error("more section not restored");
  if (!visible(tools.window.document, "moreToolsPanel")) throw new Error("tools panel not visible");

  const overview = await load("#/nonsense");
  if (overview.window.document.title !== "Overview") throw new Error("title not Overview");
  if (!visible(overview.window.document, "overviewPanel")) throw new Error("overview fallback failed");

  const more = await load("#/more");
  if (!more.window.document.getElementById("deploymentInfo")) throw new Error("deploymentInfo missing");
  if (more.window.document.getElementById("buildInfo")) throw new Error("buildInfo still present");

  console.log("UX-POLISH PASS: title, dropdown routing, retention, deep links, deployment placement");
})().catch((error) => {
  console.error("UX-POLISH FAIL:", error.message);
  process.exit(1);
});
```
